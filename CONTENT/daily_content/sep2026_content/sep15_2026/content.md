## SCIENCE NOTE — Founder Interview Prep (not for publishing)
**Ingredient / mechanism:** Taurine — osmolyte function and mitochondrial cytoprotection

Taurine (2-aminoethanesulfonic acid) is among the most abundant free amino acids in mammalian tissue, particularly concentrated in skeletal muscle, cardiac muscle, and the brain. Unlike many conditionally essential amino acids, taurine is not incorporated into proteins. Its primary functional roles are osmotic regulation and mitochondrial membrane stabilization. At the mitochondrial level, taurine forms a modified uridine derivative (5-taurinomethyluridine) within mitochondrial tRNA, which is essential for the accurate translation of mitochondrially encoded subunits of the electron transport chain. Deficiency in this modification is associated with impaired ATP synthesis efficiency and elevated reactive oxygen species output. In the central nervous system, taurine acts as a partial agonist at GABA-A receptors and glycine receptors, providing mild inhibitory tone without sedation — which may mechanistically explain the attenuation of caffeine-driven overstimulation observed in combination studies.

Power Coffee delivers 2.1g per serving, consistent with the dosing range used in human performance and cardiovascular research.

**Reference:** Schaffer SW, Ju Jong C, KC R, Azuma J. Physiological roles of taurine in heart and muscle. J Biomed Sci. 2010;17 Suppl 1:S2. PMID: 20804598 [VERIFY CITATION for mitochondrial tRNA detail — Sastre 2002 lineage; confirm most current PMID before use on air]

**Interview angle:** Taurine supports mitochondrial translation efficiency and may modulate the excitatory overstimulation associated with high caffeine intake — which is a plausible mechanism for the smoother energy profile users report.

function addRect(parent, x, y, w, h, color, opacity) {
  if (opacity === undefined) { opacity = 1; }
  var r = figma.createRectangle();
  r.x = x; r.y = y; r.resize(w, h);
  r.fills = [{ type: "SOLID", color: col(color), opacity: opacity }];
  parent.appendChild(r);
  return r;
}

// Font map — matches make_carousel.py (Pillow) rendering
// Headlines/body → Georgia | Labels/pills/footer → Arial
function _fontFor(weight) {
  if (weight === "Black Italic" || weight === "Bold Italic") { return { family: "Georgia", style: "Bold Italic" }; }
  if (weight === "Black" || weight === "Bold" && false)      { return { family: "Georgia", style: "Bold" }; }
  if (weight === "Italic" || weight === "Regular Italic")    { return { family: "Georgia", style: "Italic" }; }
  // Labels, overlines, pills, footer, counter use Arial
  if (weight === "Bold")    { return { family: "Arial", style: "Bold" }; }
  if (weight === "Regular") { return { family: "Georgia", style: "Regular" }; }
  return { family: "Arial", style: weight };
}

async function addText(parent, txt, x, y, w, size, weight, color, align, lineH) {
  if (align === undefined) { align = "LEFT"; }
  if (lineH === undefined) { lineH = 1.05; }
  var fn = _fontFor(weight);
  await figma.loadFontAsync(fn);
  var t = figma.createText();
  t.fontName = fn;
  t.characters = txt;
  t.fontSize = size;
  t.textAlignHorizontal = align;
  t.fills = [{ type: "SOLID", color: col(color) }];
  t.lineHeight = { unit: "PERCENT", value: lineH * 100 };
  t.x = x; t.y = y;
  t.resize(w, t.height);
  parent.appendChild(t);
  return t;
}
