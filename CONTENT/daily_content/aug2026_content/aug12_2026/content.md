### Body

## The Afternoon Is Where Most People Quietly Give Up

You started the day strong. By early afternoon you were grinding through tasks that should take thirty minutes, refreshing your inbox, and reaching for something — anything — to feel sharp again.

If that pattern sounds familiar, you are not alone. And you are not lazy.

## The Real Reason Your Energy Falls Apart at 2pm

Most people assume the afternoon crash is about sleep, stress, or willpower. Sometimes it is. But more often it comes down to a simple design flaw in how we fuel the morning.

Standard coffee gives your system one tool: caffeine. Caffeine works by blocking adenosine receptors — the signals your brain uses to register fatigue. For the first two to four hours, that blocking effect is strong. Then it starts to break down. Adenosine floods back in. Cortisol drops. And suddenly you are sitting in the same chair with the same responsibilities, but a completely different nervous system.

## Why a Third Cup Does Not Solve the Problem

Adding more caffeine at noon may buy you forty-five minutes. It may also cost you your evening — elevated heart rate, shallow sleep, starting tomorrow in deficit.

The problem was never the quantity of caffeine. It was the absence of the compounds that support your system through the back half of the curve.

## What Actually Supports Steady Energy Through the Afternoon

Ingredients like taurine may help buffer the nervous system during and after peak caffeine activity. Ginkgo biloba has been studied for its role in supporting circulation and mental clarity. Thermogenic spices like cinnamon support steady blood sugar — one of the quieter drivers of energy stability across the day.

Together, these compounds do not replace your coffee. They make it work longer and leave you without the floor-drop most people have accepted as normal.

## SCIENCE NOTE — Founder Interview Prep (not for publishing)
**Ingredient / mechanism:** Taurine — osmolyte function and neuromodulatory action on GABA-A receptors

Taurine (2-aminoethanesulfonic acid) is the most abundant free amino acid in excitable tissues — brain, heart, skeletal muscle. It is not incorporated into proteins; instead it functions as a cytoprotective osmolyte, regulating cell volume and membrane potential under metabolic stress. Its neuromodulatory role is mediated in part through agonism at GABA-A and glycine receptors, which reduces excessive neuronal excitability without producing sedation. This mechanism is distinct from classic GABAergic sedatives: taurine's effect is modulatory and concentration-dependent, dampening hyperexcitability at peak stimulant exposure rather than globally suppressing CNS activity. In the context of co-administration with caffeine, taurine may attenuate caffeine's cortisol-amplifying and sympathomimetic effects by limiting excessive adrenergic signaling downstream of adenosine receptor blockade. Power Coffee delivers 2.1g per serving — a dose consistent with ranges studied in endurance and cognitive performance contexts.

**Reference:** Ripps H, Shen W. "Review: Taurine: A very essential amino acid." Molecular Vision. 2012;18:2673-2686. PMID: 23170060 [VERIFY CITATION for taurine/GABA-A receptor specificity language]

**Interview angle:** Taurine at this dose may support nervous system stability during peak caffeine activity — modulating excitability rather than adding to it, which is a meaningfully different mechanism than simply stacking more stimulants.

// BRAND COLORS - use quoted string keys in SLIDES data
const COLORS = {
  BLACK:   { r: 0.07, g: 0.06, b: 0.05 },
  BROWN:   { r: 0.48, g: 0.24, b: 0.12 },
  BROWN_L: { r: 0.75, g: 0.48, b: 0.28 },
  CREAM:   { r: 0.94, g: 0.91, b: 0.84 },
  WHITE:   { r: 1,    g: 1,    b: 1    },
  DARK_BG: { r: 0.10, g: 0.09, b: 0.08 },
};
function col(name) { return COLORS[name] || name; }

const W = 1080;
const H = 1920;
const GAP = 60;

// SLIDE DATA - accent/bg/textColor must be quoted strings: "BLACK", "BROWN", "BROWN_L", "CREAM", "WHITE"
const SLIDES = [{ id: 1, slideType: "hook", bg: "BLACK", accent: "BROWN", textColor: "CREAM", overline: "THE AFTERNOON PROBLEM", headline: "YOU LOST\nTHE AFTERNOON\nAGAIN", subtext: "It is not willpower. It is what your coffee is missing." }
];

// ── HELPERS ──────────────────────────────────────────────────
function addRect(parent, x, y, w, h, color, opacity) {
  if (opacity === undefined) { opacity = 1; }
  var r = figma.createRectangle();
  r.x = x; r.y = y; r.resize(w, h);
  r.fills = [{ type: "SOLID", color: col(color), opacity: opacity }];
  parent.appendChild(r);
  return r;
}

// Font map — matches make_carousel.py (Pillow) rendering
// Headlines/body → Georgia | Labels/pills/footer → Arial
function _fontFor(weight) {
  if (weight === "Black Italic" || weight === "Bold Italic") { return { family: "Georgia", style: "Bold Italic" }; }
  if (weight === "Black" || weight === "Bold" && false)      { return { family: "Georgia", style: "Bold" }; }
  if (weight === "Italic" || weight === "Regular Italic")    { return { family: "Georgia", style: "Italic" }; }
  // Labels, overlines, pills, footer, counter use Arial
  if (weight === "Bold")    { return { family: "Arial", style: "Bold" }; }
  if (weight === "Regular") { return { family: "Georgia", style: "Regular" }; }
  return { family: "Arial", style: weight };
}

async function addText(parent, txt, x, y, w, size, weight, color, align, lineH) {
  if (align === undefined) { align = "LEFT"; }
  if (lineH === undefined) { lineH = 1.05; }
  var fn = _fontFor(weight);
  await figma.loadFontAsync(fn);
  var t = figma.createText();
  t.fontName = fn;
  t.characters = txt;
  t.fontSize = size;
  t.textAlignHorizontal = align;
  t.fills = [{ type: "SOLID", color: col(color) }];
  t.lineHeight = { unit: "PERCENT", value: lineH * 100 };
  t.x = x; t.y = y;
  t.resize(w, t.height);
  parent.appendChild(t);
  return t;
}

async function buildSlide(data, offsetX) {
  var frame = figma.createFrame();
  frame.name = "Slide " + data.id + " - " + data.slideType;
  frame.resize(W, H);
  frame.x = offsetX; frame.y = 0;
  frame.fills = [{ type: "SOLID", color: col(data.bg) }];
  frame.clipsContent = true;

  for (var i = 0; i < 12; i++) {
    var line = figma.createLine();
    line.x = -100 + i * 110; line.y = 0;
    line.resize(H * 1.5, 0);
    line.rotation = -55;
    line.strokes = [{ type: "SOLID", color: col(data.accent), opacity: 0.04 }];
    line.strokeWeight = 40;
    frame.appendChild(line);
  }

  await addText(frame, "0" + data.id, 64, 60, 120, 13, "Bold", "WHITE");
  addRect(frame, 64, 85, 20, 2, data.accent);

  if (data.overline) {
    await addText(frame, data.overline, 64, 96, W - 128, 12, "Bold", "BROWN_L", "LEFT", 1.4);
  }

  if (data.pivotBar) { addRect(frame, 64, 148, 6, 260, data.accent); }

  var hX; if (data.pivotBar) { hX = 90; } else { hX = 64; }
  var hlColor; if (data.isCTA) { hlColor = "WHITE"; } else { hlColor = data.textColor; }
  await addText(frame, data.headline, hX, 148, W - hX - 40, 128, "Black Italic", hlColor, "LEFT", 0.92);

  var divY; if (data.isCTA) { divY = 740; } else { divY = 730; }
  addRect(frame, 64, divY, W - 128, 2, data.accent, 0.6);

  if (data.pills) {
    var pillX = 64;
    for (var j = 0; j < data.pills.length; j++) {
      var pill = data.pills[j];
      await figma.loadFontAsync({ family: "Arial", style: "Bold" });
      var pf = figma.createFrame();
      pf.resize(160, 36); pf.x = pillX; pf.y = divY + 20;
      pf.fills = [{ type: "SOLID", color: col("BROWN"), opacity: 0.25 }];
      pf.cornerRadius = 0;
      frame.appendChild(pf);
      var pt = figma.createText();
      pt.fontName = { family: "Arial", style: "Bold" };
      pt.characters = pill; pt.fontSize = 10;
      pt.textAlignHorizontal = "CENTER";
      pt.fills = [{ type: "SOLID", color: col("CREAM") }];
      pt.letterSpacing = { unit: "PERCENT", value: 20 };
      pt.resize(160, 36); pt.x = 0; pt.y = 12;
      pf.appendChild(pt);
      pillX += 172;
    }
  }

  if (data.subtext) {
    var subY; if (data.pills) { subY = divY + 70; } else { subY = divY + 24; }
    await addText(frame, data.subtext, 64, subY, W - 128, 22, "Regular", { r: 0.85, g: 0.80, b: 0.72 }, "LEFT", 1.6);
  }

  addRect(frame, 64, H - 96, W - 128, 1, "BROWN_L", 0.3);
  await addText(frame, "THE POWER COFFEE", 64, H - 78, 400, 11, "Bold", "BROWN_L", "LEFT", 1.4);
  await addText(frame, "CLEAN ENERGY. REAL FOCUS.", W - 380, H - 78, 316, 10, "Regular", { r: 0.6, g: 0.55, b: 0.48 }, "RIGHT", 1.4);

  if (data.isCTA) {
    addRect(frame, 0, H - 180, W, 180, "BROWN");
    await addText(frame, "POWER COFFEE", 64, H - 138, W - 128, 14, "Black", "WHITE", "LEFT", 1.2);
    await addText(frame, "thepowercoffee.com", 64, H - 110, W - 128, 13, "Regular", { r: 1, g: 0.95, b: 0.88 }, "LEFT", 1.4);
    await addText(frame, "@POWERCOFFEE.OFC", 64, H - 76, W - 128, 12, "Bold", "WHITE", "LEFT", 1.4);
  }

  figma.currentPage.appendChild(frame);
  return frame;
}

async function main() {
  var frames = [];
  for (var i = 0; i < SLIDES.length; i++) {
    var frame = await buildSlide(SLIDES[i], i * (W + GAP));
    frames.push(frame);
  }
  figma.viewport.scrollAndZoomIntoView(frames);
  figma.notify("Power Coffee - slides created!", { timeout: 4000 });
}

main();
