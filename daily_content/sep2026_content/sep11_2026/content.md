## SCIENCE NOTE — Founder Interview Prep (not for publishing)
**Ingredient / mechanism:** Taurine as an osmolyte and cytoprotective modulator in sustained-effort contexts

Taurine is the most abundant free amino acid in excitable tissues — heart, skeletal muscle, and brain. Its primary mechanistic role is not stimulant or ergogenic in the classical sense. Taurine functions as an organic osmolyte, stabilizing cell volume under conditions of metabolic stress by regulating intracellular calcium handling and membrane potential. In neural tissue, it acts as an agonist at glycine receptors and a partial agonist at GABA-A receptors, moderating excitatory tone without inducing sedation. This is mechanistically distinct from caffeine's adenosine antagonism. When caffeine drives increased neuronal firing rate, taurine appears to buffer the downstream excitatory overshoot — the physiological correlate of what users describe as "smooth" energy versus jitter. Power Coffee delivers 2.1g per 10g serving, within the range studied for exercise performance and cellular protection. EFSA has reviewed taurine at intakes up to 6g/day as safe in healthy adults.

**Reference:** Schaffer S, Kim HW. Effects and Mechanisms of Taurine as a Therapeutic Agent. Biomol Ther (Seoul). 2018;26(3):225-241. PMID: 29631391

**Interview angle:** Taurine does not accelerate the nervous system — it supports the conditions under which the nervous system can sustain effort without the excitatory overshoot that produces jitters and the eventual energy cliff.

// ── HELPERS ──────────────────────────────────────────────────
function addRect(parent, x, y, w, h, color, opacity) {
  if (opacity === undefined) { opacity = 1; }
  var r = figma.createRectangle();
  r.x = x; r.y = y; r.resize(w, h);
  r.fills = [{ type: "SOLID", color: col(color), opacity: opacity }];
  parent.appendChild(r);
  return r;
}

// Font map — matches make_carousel.py (Pillow) rendering
// Headlines/body → Georgia | Labels/pills/footer → Arial
function _fontFor(weight) {
  if (weight === "Black Italic" || weight === "Bold Italic") { return { family: "Georgia", style: "Bold Italic" }; }
  if (weight === "Black" || weight === "Bold" && false)      { return { family: "Georgia", style: "Bold" }; }
  if (weight === "Italic" || weight === "Regular Italic")    { return { family: "Georgia", style: "Italic" }; }
  // Labels, overlines, pills, footer, counter use Arial
  if (weight === "Bold")    { return { family: "Arial", style: "Bold" }; }
  if (weight === "Regular") { return { family: "Georgia", style: "Regular" }; }
  return { family: "Arial", style: weight };
}

async function addText(parent, txt, x, y, w, size, weight, color, align, lineH) {
  if (align === undefined) { align = "LEFT"; }
  if (lineH === undefined) { lineH = 1.05; }
  var fn = _fontFor(weight);
  await figma.loadFontAsync(fn);
  var t = figma.createText();
  t.fontName = fn;
  t.characters = txt;
  t.fontSize = size;
  t.textAlignHorizontal = align;
  t.fills = [{ type: "SOLID", color: col(color) }];
  t.lineHeight = { unit: "PERCENT", value: lineH * 100 };
  t.x = x; t.y = y;
  t.resize(w, t.height);
  parent.appendChild(t);
  return t;
}

async function buildSlide(data, offsetX) {

  frame.clipsContent = true;

    line.x = -100 + i * 110; line.y = 0;
    line.resize(H * 1.5, 0);
    line.rotation = -55;
    line.strokes = [{ type: "SOLID", color: col(data.accent), opacity: 0.04 }];
    line.strokeWeight = 40;

  var hX; if (data.pivotBar) { hX = 90; } else { hX = 64; }
  var hlColor; if (data.isCTA) { hlColor = "WHITE"; } else { hlColor = data.textColor; }

  if (data.subtext) {

}
