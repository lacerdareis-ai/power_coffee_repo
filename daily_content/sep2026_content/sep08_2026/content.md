## SCIENCE NOTE — Founder Interview Prep (not for publishing)
**Ingredient / mechanism:** Taurine — osmolytic and cytoprotective action under neurological demand

Taurine (2-aminoethanesulfonic acid) is conditionally essential, meaning endogenous synthesis via cysteine sulfinic acid decarboxylase is adequate at baseline but insufficient under sustained oxidative or metabolic stress. At the neuronal level, taurine functions as an osmolyte — regulating intracellular volume and chloride homeostasis — and as an endogenous ligand at glycine and GABA-A receptors, producing mild inhibitory modulation without sedation. This is mechanistically distinct from classical inhibitory neurotransmitters: taurine appears to act as a partial agonist, dampening excitotoxic overstimulation rather than globally suppressing neural activity. Under conditions of high cognitive or physical demand, intracellular taurine in neurons and muscle depletes measurably, and animal models demonstrate impaired mitochondrial respiratory chain efficiency with low taurine status. Power Coffee delivers 2.1g per serving, consistent with ranges used in human performance studies. Each serving's caffeine content (~175mg natural caffeine) sits at or below the EFSA-established single-dose safety threshold of 200mg.

**Reference:** Ripps H, Shen W. "Review: taurine: a 'very essential' amino acid." Molecular Vision, 2012;18:2673-2686. PMID: 23170060 [VERIFY CITATION for specific mechanistic claims on GABA-A partial agonism]

**Interview angle:** Taurine may help buffer the overstimulation that high-caffeine states can produce — supporting a steadier neurological baseline rather than simply adding more stimulatory signal.

function addRect(parent, x, y, w, h, color, opacity) {
  if (opacity === undefined) { opacity = 1; }
  var r = figma.createRectangle();
  r.x = x; r.y = y; r.resize(w, h);
  r.fills = [{ type: "SOLID", color: col(color), opacity: opacity }];
  parent.appendChild(r);
  return r;

// Font map — matches make_carousel.py (Pillow) rendering
// Headlines/body → Georgia | Labels/pills/footer → Arial
function _fontFor(weight) {
  if (weight === "Black Italic" || weight === "Bold Italic") { return { family: "Georgia", style: "Bold Italic" }; }
  if (weight === "Black" || weight === "Bold" && false)      { return { family: "Georgia", style: "Bold" }; }
  if (weight === "Italic" || weight === "Regular Italic")    { return { family: "Georgia", style: "Italic" }; }
  // Labels, overlines, pills, footer, counter use Arial
  if (weight === "Bold")    { return { family: "Arial", style: "Bold" }; }
  if (weight === "Regular") { return { family: "Georgia", style: "Regular" }; }
  return { family: "Arial", style: weight };
}

async function addText(parent, txt, x, y, w, size, weight, color, align, lineH) {
  if (align === undefined) { align = "LEFT"; }
  if (lineH === undefined) { lineH = 1.05; }
  var fn = _fontFor(weight);
  await figma.loadFontAsync(fn);
  var t = figma.createText();
  t.fontName = fn;
  t.characters = txt;
  t.fontSize = size;
  t.textAlignHorizontal = align;
  t.fills = [{ type: "SOLID", color: col(color) }];
  t.lineHeight = { unit: "PERCENT", value: lineH * 100 };
  t.x = x; t.y = y;
  t.resize(w, t.height);
  parent.appendChild(t);
  return t;
}
