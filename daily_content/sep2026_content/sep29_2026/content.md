## SCIENCE NOTE — Founder Interview Prep (not for publishing)
**Ingredient / mechanism:** Taurine — osmolytic and neuroregulatory function under caffeine load

Taurine (2-aminoethanesulfonic acid) is one of the most abundant free amino acids in human neural tissue, concentrated especially in the cerebellum, hippocampus, and retina. Unlike stimulatory amino acids, taurine acts primarily as an inhibitory neuromodulator: it is a partial agonist at GABA-A receptors and a direct agonist at glycine receptors, both inhibitory ion channels. Under conditions of high sympathetic activation — such as a large acute caffeine dose — endogenous taurine is depleted more rapidly through urinary excretion and elevated cellular demand. Exogenous supplementation at doses of 1–2.1g has been shown to attenuate caffeine-induced anxiogenic effects and blunt excessive sympathetic tone without sedation, supporting what users experience as "smooth" versus "spiky" energy. The mechanism is not sedative; it is regulatory — restoring signal balance rather than suppressing arousal. Power Coffee's 2.1g dose per 10g serving sits at the upper boundary of what peer-reviewed supplementation studies have examined for acute neurological effect.

**Reference:** Geidl-Flueck B et al., and supporting data in Shao A & Hathcock JN, Regulatory Toxicology and Pharmacology, 2008; 46(1):1-27. [VERIFY CITATION for the caffeine-taurine anxiolysis mechanism specifically]

**Interview angle:** Taurine at this dose may help regulate the nervous system response to caffeine — supporting steadier energy rather than amplifying the stimulant load.

function addRect(parent, x, y, w, h, color, opacity) {
  if (opacity === undefined) { opacity = 1; }
  var r = figma.createRectangle();
  r.x = x; r.y = y; r.resize(w, h);
  r.fills = [{ type: "SOLID", color: col(color), opacity: opacity }];
  parent.appendChild(r);
  return r;
}

// Font map — matches make_carousel.py (Pillow) rendering
// Headlines/body → Georgia | Labels/pills/footer → Arial
function _fontFor(weight) {
  if (weight === "Black Italic" || weight === "Bold Italic") { return { family: "Georgia", style: "Bold Italic" }; }
  if (weight === "Black" || weight === "Bold" && false)      { return { family: "Georgia", style: "Bold" }; }
  if (weight === "Italic" || weight === "Regular Italic")    { return { family: "Georgia", style: "Italic" }; }
  // Labels, overlines, pills, footer, counter use Arial
  if (weight === "Bold")    { return { family: "Arial", style: "Bold" }; }
  if (weight === "Regular") { return { family: "Georgia", style: "Regular" }; }
  return { family: "Arial", style: weight };
}

async function addText(parent, txt, x, y, w, size, weight, color, align, lineH) {
  if (align === undefined) { align = "LEFT"; }
  if (lineH === undefined) { lineH = 1.05; }
  var fn = _fontFor(weight);
  await figma.loadFontAsync(fn);
  var t = figma.createText();
  t.fontName = fn;
  t.characters = txt;
  t.fontSize = size;
  t.textAlignHorizontal = align;
  t.fills = [{ type: "SOLID", color: col(color) }];
  t.lineHeight = { unit: "PERCENT", value: lineH * 100 };
  t.x = x; t.y = y;
  t.resize(w, t.height);
  parent.appendChild(t);
  return t;
}
