## SCIENCE NOTE — Founder Interview Prep (not for publishing)
**Ingredient / mechanism:** Taurine — osmolyte function and neurological stabilization under caffeine co-administration

Taurine (2-aminoethanesulfonic acid) is conditionally essential under physiological stress — synthesized endogenously from cysteine via the cysteine sulfinic acid pathway but depleted rapidly during sustained cognitive or physical load. Its primary neurological role is as an endogenous agonist at GABA-A receptors and glycine receptors in the central nervous system, producing an inhibitory tone that counteracts the excitatory pressure that high-dose caffeine can generate via adenosine A1 and A2A receptor blockade. Taurine also functions as a cellular osmolyte, regulating intracellular calcium handling and protecting mitochondrial membrane integrity under oxidative stress — a mechanism distinct from its GABAergic activity. In the context of co-administration with caffeine at the doses present in Power Coffee (natural caffeine approximately 175mg, taurine 2.1g — within EFSA's acknowledged safe single-dose range), the theoretical effect is modulation of caffeine's excitatory ceiling without blunting wakefulness, which corresponds to the reported subjective experience of "clean energy."

**Reference:** Ripps H, Shen W. "Review: taurine: a 'very essential' amino acid." Molecular Vision, 2012;18:2673-2686. PMID: 23170060 [VERIFY for specific GABAergic receptor agonism claims — core osmolyte and conditional essentiality findings are well-established in this review.]

**Interview angle:** Taurine in Power Coffee may support the stability of the caffeine response by modulating inhibitory receptor tone — it does not amplify stimulation, it may help contain it.

function addRect(parent, x, y, w, h, color, opacity) {
  if (opacity === undefined) { opacity = 1; }
  var r = figma.createRectangle();
  r.x = x; r.y = y; r.resize(w, h);
  r.fills = [{ type: "SOLID", color: col(color), opacity: opacity }];
  parent.appendChild(r);
  return r;

// Font map — matches make_carousel.py (Pillow) rendering
// Headlines/body → Georgia | Labels/pills/footer → Arial
function _fontFor(weight) {
  if (weight === "Black Italic" || weight === "Bold Italic") { return { family: "Georgia", style: "Bold Italic" }; }
  if (weight === "Black" || weight === "Bold" && false)      { return { family: "Georgia", style: "Bold" }; }
  if (weight === "Italic" || weight === "Regular Italic")    { return { family: "Georgia", style: "Italic" }; }
  // Labels, overlines, pills, footer, counter use Arial
  if (weight === "Bold")    { return { family: "Arial", style: "Bold" }; }
  if (weight === "Regular") { return { family: "Georgia", style: "Regular" }; }
  return { family: "Arial", style: weight };
}

async function addText(parent, txt, x, y, w, size, weight, color, align, lineH) {
  if (align === undefined) { align = "LEFT"; }
  if (lineH === undefined) { lineH = 1.05; }
  var fn = _fontFor(weight);
  await figma.loadFontAsync(fn);
  var t = figma.createText();
  t.fontName = fn;
  t.characters = txt;
  t.fontSize = size;
  t.textAlignHorizontal = align;
  t.fills = [{ type: "SOLID", color: col(color) }];
  t.lineHeight = { unit: "PERCENT", value: lineH * 100 };
  t.x = x; t.y = y;
  t.resize(w, t.height);
  parent.appendChild(t);
  return t;
}
