## The 11AM Crash Has Nothing to Do With Caffeine

If you've ever had two cups of coffee and still hit a wall by mid-morning, the problem isn't your tolerance to caffeine. The problem is blood sugar — and most coffee does nothing to manage it.

Regular coffee spikes cortisol. Without anything to buffer the glucose response, you get a sharp energy peak followed by a hard drop. That's the crash. It's physiological, and it's predictable.

## What Pea Protein Actually Does

Pea protein is a plant-based protein with a complete amino acid profile. When consumed alongside caffeine, it slows glucose absorption, which flattens the post-coffee blood sugar spike and extends the energy release window. The result isn't a muted coffee experience — it's a steadier, longer-lasting one.

Studies on protein and caffeine co-ingestion show measurable improvements in sustained cognitive performance versus caffeine alone. Your brain gets the focus signal, and your body has something to hold it in place.

## Why Pea Protein Specifically

Whey works — but not for everyone, every day. Pea protein is lactose-free and digests cleanly, making it a better daily-use option. It has no chalky aftertaste, no bloating, and a full amino acid profile that supports both cognitive function and muscle recovery.

In a coffee context, that matters. Your morning ritual should fuel you, not distract you.

// ── HELPERS ──────────────────────────────────────────────────
function addRect(parent, x, y, w, h, color, opacity) {
  if (opacity === undefined) { opacity = 1; }
  var r = figma.createRectangle();
  r.x = x; r.y = y; r.resize(w, h);
  r.fills = [{ type: "SOLID", color: col(color), opacity: opacity }];
  parent.appendChild(r);
  return r;
}

async function addText(parent, txt, x, y, w, size, weight, color, align, lineH) {
  if (align === undefined) { align = "LEFT"; }
  if (lineH === undefined) { lineH = 1.05; }
  await figma.loadFontAsync({ family: "Inter", style: weight });
  var t = figma.createText();
  t.fontName = { family: "Inter", style: weight };
  t.characters = txt;
  t.fontSize = size;
  t.textAlignHorizontal = align;
  t.fills = [{ type: "SOLID", color: col(color) }];
  t.lineHeight = { unit: "PERCENT", value: lineH * 100 };
  t.x = x; t.y = y;
  t.resize(w, t.height);
  parent.appendChild(t);
  return t;
}

async function buildSlide(data, offsetX) {
  var frame = figma.createFrame();
  frame.name = "Slide " + data.id + " - " + data.slideType;
  frame.resize(W, H);
  frame.x = offsetX; frame.y = 0;
  frame.clipsContent = true;

  // Background: CREAM for content slides, BLACK for CTA
  var bgColor; if (data.isCTA) { bgColor = "BLACK"; } else { bgColor = "CREAM"; }
  frame.fills = [{ type: "SOLID", color: col(bgColor) }];

  // Faint watermark slide number (content slides only)
  if (!data.isCTA) {
    await figma.loadFontAsync({ family: "Inter", style: "Black" });
    var wmNum; if (data.id < 10) { wmNum = "0" + data.id; } else { wmNum = "" + data.id; }
    var wm = figma.createText();
    wm.fontName = { family: "Inter", style: "Black" };
    wm.characters = wmNum;
    wm.fontSize = 360;
    wm.fills = [{ type: "SOLID", color: col("BROWN"), opacity: 0.07 }];
    wm.x = 580; wm.y = 620;
    frame.appendChild(wm);
  }

  // BROWN top bar (108px)
  addRect(frame, 0, 0, W, 108, "BROWN");

  // Overline inside top bar
  if (data.overline) {
    await addText(frame, data.overline, 64, 42, W - 250, 12, "Bold", "WHITE", "LEFT", 1.4);
  }

  // Slide counter inside top bar (right-aligned)
  var idStr; if (data.id < 10) { idStr = "0" + data.id; } else { idStr = "" + data.id; }
  var totalStr; if (TOTAL < 10) { totalStr = "0" + TOTAL; } else { totalStr = "" + TOTAL; }
  var countStr = idStr + " / " + totalStr;
  await addText(frame, countStr, W - 190, 42, 126, 11, "Regular", { r: 0.80, g: 0.77, b: 0.70 }, "RIGHT", 1.4);

  // 6px vertical BROWN accent bar
  addRect(frame, 64, 148, 6, 340, "BROWN");

  // Headline: 92px Black (not italic)
  var hlColor; if (data.isCTA) { hlColor = "CREAM"; } else { hlColor = "BLACK"; }
  await addText(frame, data.headline, 90, 148, W - 154, 92, "Black", hlColor, "LEFT", 0.90);

  // Horizontal divider
  addRect(frame, 64, 730, W - 128, 2, "BROWN_L", 0.5);

  // Pills (if any)
  if (data.pills) {
    var pillX = 64;
    for (var j = 0; j < data.pills.length; j++) {
      var pill = data.pills[j];
      await figma.loadFontAsync({ family: "Inter", style: "Bold" });
      var pf = figma.createFrame();
      pf.resize(160, 36); pf.x = pillX; pf.y = 752;
      pf.fills = [{ type: "SOLID", color: col("BROWN"), opacity: 0.15 }];
      pf.cornerRadius = 4;
      frame.appendChild(pf);
      var pt = figma.createText();
      pt.fontName = { family: "Inter", style: "Bold" };
      pt.characters = pill; pt.fontSize = 10;
      pt.textAlignHorizontal = "CENTER";
      pt.fills = [{ type: "SOLID", color: col("BROWN") }];
      pt.letterSpacing = { unit: "PERCENT", value: 20 };
      pt.resize(160, 36); pt.x = 0; pt.y = 12;
      pf.appendChild(pt);
      pillX += 172;
    }
  }

  // Subtext
  if (data.subtext) {
    var subY; if (data.pills) { subY = 800; } else { subY = 752; }
    var subColor;
    if (data.isCTA) { subColor = { r: 0.90, g: 0.87, b: 0.80 }; } else { subColor = { r: 0.25, g: 0.18, b: 0.10 }; }
    await addText(frame, data.subtext, 64, subY, W - 128, 22, "Regular", subColor, "LEFT", 1.6);
  }

  // Footer
  if (data.isCTA) {
    addRect(frame, 0, H - 156, W, 156, "BROWN");
    await addText(frame, "POWER COFFEE", 64, H - 126, W - 128, 14, "Black", "WHITE", "LEFT", 1.2);
    await addText(frame, "thepowercoffee.com", 64, H - 98, W - 128, 13, "Regular", { r: 0.94, g: 0.91, b: 0.84 }, "LEFT", 1.4);
    await addText(frame, "@powercoffee.ofc", 64, H - 60, W - 128, 12, "Bold", "WHITE", "LEFT", 1.4);
  } else {
    addRect(frame, 64, H - 64, W - 128, 1, "BROWN_L", 0.4);
    await addText(frame, "THE POWER COFFEE", 64, H - 52, 400, 10, "Bold", "BROWN", "LEFT", 1.4);

  }

  figma.currentPage.appendChild(frame);
  return frame;
}

async function main() {
  var frames = [];
  for (var i = 0; i < SLIDES.length; i++) {
    var frame = await buildSlide(SLIDES[i], i * (W + GAP));
    frames.push(frame);
  }
  figma.viewport.scrollAndZoomIntoView(frames);
  figma.notify("Power Coffee - slides created!", { timeout: 4000 });
}

main();
