### Body

## The Problem With Caffeine on Its Own

Caffeine is effective. No one disputes that. But it works by blocking adenosine — the neurotransmitter that signals tiredness — while simultaneously triggering a cortisol response. For many people, that means energy that arrives fast, feels anxious, and disappears before lunch.

This isn't a caffeine problem. It's a caffeine-alone problem.

## What Matcha Actually Contains

Matcha is ground whole-leaf green tea. Unlike steeped green tea, matcha delivers the entire leaf's compounds in every serving — including a meaningful dose of L-theanine, an amino acid that occurs naturally in the tea plant.

L-theanine promotes alpha brain wave activity. It modulates the stimulation of caffeine without suppressing it. In practical terms: you stay alert, but the edge comes off. Focus sharpens. The scatter goes away.

## What the Research Shows

The combination of L-theanine and caffeine has been studied for over 15 years. Findings consistently show improvements in sustained attention, reaction time, and working memory — performing better than caffeine alone across multiple cognitive tasks.

Matcha also releases its caffeine more gradually than coffee, over 4–6 hours rather than the sharp spike-and-drop curve most people recognize from their morning cup.

## Why Dose Matters

Many functional coffee brands include matcha on their ingredient list. Fewer include it at a dose that does anything measurable. Therapeutic effects are observed starting around 200mg. If a product doesn't tell you how much matcha it contains, that's worth asking.

function addRect(parent, x, y, w, h, color, opacity) {
  if (opacity === undefined) { opacity = 1; }
  var r = figma.createRectangle();
  r.x = x; r.y = y; r.resize(w, h);
  r.fills = [{ type: "SOLID", color: col(color), opacity: opacity }];
  parent.appendChild(r);
  return r;
}

// Font map — matches make_carousel.py (Pillow) rendering
// Headlines/body → Georgia | Labels/pills/footer → Arial
function _fontFor(weight) {
  if (weight === "Black Italic" || weight === "Bold Italic") { return { family: "Georgia", style: "Bold Italic" }; }
  if (weight === "Black" || weight === "Bold" && false)      { return { family: "Georgia", style: "Bold" }; }
  if (weight === "Italic" || weight === "Regular Italic")    { return { family: "Georgia", style: "Italic" }; }
  // Labels, overlines, pills, footer, counter use Arial
  if (weight === "Bold")    { return { family: "Arial", style: "Bold" }; }
  if (weight === "Regular") { return { family: "Georgia", style: "Regular" }; }
  return { family: "Arial", style: weight };
}

async function addText(parent, txt, x, y, w, size, weight, color, align, lineH) {
  if (align === undefined) { align = "LEFT"; }
  if (lineH === undefined) { lineH = 1.05; }
  var fn = _fontFor(weight);
  await figma.loadFontAsync(fn);
  var t = figma.createText();
  t.fontName = fn;
  t.characters = txt;
  t.fontSize = size;
  t.textAlignHorizontal = align;
  t.fills = [{ type: "SOLID", color: col(color) }];
  t.lineHeight = { unit: "PERCENT", value: lineH * 100 };
  t.x = x; t.y = y;
  t.resize(w, t.height);
  parent.appendChild(t);
  return t;
}
