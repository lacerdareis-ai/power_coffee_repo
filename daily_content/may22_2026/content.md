<h2>The Blood Sugar Problem Nobody Talks About</h2>

<p>Most conversations about morning energy focus on caffeine — how much, when, how fast it hits. But the reason most people still feel the 11am slump has less to do with caffeine and more to do with blood sugar.</p>

<p>When you drink black coffee on an empty stomach, a cortisol response temporarily raises blood glucose — then drops it. That drop is what most people call the "coffee crash." It's not the caffeine wearing off. It's your body reacting to a glucose fluctuation it wasn't prepared for.</p>

<h2>Why Cinnamon Changes the Equation</h2>

<p>Cinnamon has a well-documented research history as a blood sugar stabilizer. Studies suggest it may support insulin sensitivity and slow the rate at which glucose enters the bloodstream. The practical effect: when blood sugar is stable, energy is stable. No spike. No drop. No slump at 10am.</p>

<p>Every serving of Power Coffee includes 1.32g of cinnamon — not as a flavoring, but as a functional ingredient. It works alongside matcha, pea protein, and taurine in our formula to support a clean, sustained energy arc from your first scoop through your last morning meeting.</p>

function addRect(parent, x, y, w, h, color, opacity) {
  if (opacity === undefined) { opacity = 1; }
  var r = figma.createRectangle();
  r.x = x; r.y = y; r.resize(w, h);
  r.fills = [{ type: "SOLID", color: col(color), opacity: opacity }];
  parent.appendChild(r);
  return r;
}

async function addText(parent, txt, x, y, w, size, weight, color, align, lineH) {
  if (align === undefined) { align = "LEFT"; }
  if (lineH === undefined) { lineH = 1.05; }
  await figma.loadFontAsync({ family: "Inter", style: weight });
  var t = figma.createText();
  t.fontName = { family: "Inter", style: weight };
  t.characters = txt;
  t.fontSize = size;
  t.textAlignHorizontal = align;
  t.fills = [{ type: "SOLID", color: col(color) }];
  t.lineHeight = { unit: "PERCENT", value: lineH * 100 };
  t.x = x; t.y = y;
  t.resize(w, t.height);
  parent.appendChild(t);
  return t;
}

async function buildSlide(data, offsetX) {
  var frame = figma.createFrame();
  frame.name = "Slide " + data.id + " - " + data.slideType;
  frame.resize(W, H);
  frame.x = offsetX; frame.y = 0;
  frame.clipsContent = true;

  // Background: CREAM for content slides, BLACK for CTA
  var bgColor; if (data.isCTA) { bgColor = "BLACK"; } else { bgColor = "CREAM"; }
  frame.fills = [{ type: "SOLID", color: col(bgColor) }];

  // Faint watermark slide number (content slides only)
  if (!data.isCTA) {
    await figma.loadFontAsync({ family: "Inter", style: "Black" });
    var wmNum; if (data.id < 10) { wmNum = "0" + data.id; } else { wmNum = "" + data.id; }
    var wm = figma.createText();
    wm.fontName = { family: "Inter", style: "Black" };
    wm.characters = wmNum;
    wm.fontSize = 360;
    wm.fills = [{ type: "SOLID", color: col("BROWN"), opacity: 0.07 }];
    wm.x = 580; wm.y = 620;
    frame.appendChild(wm);
  }

  // BROWN top bar (108px)
  addRect(frame, 0, 0, W, 108, "BROWN");

  // Overline inside top bar
  if (data.overline) {
    await addText(frame, data.overline, 64, 42, W - 250, 12, "Bold", "WHITE", "LEFT", 1.4);
  }

  // Slide counter inside top bar (right-aligned)
  var idStr; if (data.id < 10) { idStr = "0" + data.id; } else { idStr = "" + data.id; }
  var totalStr; if (TOTAL < 10) { totalStr = "0" + TOTAL; } else { totalStr = "" + TOTAL; }

  // 6px vertical BROWN accent bar
  addRect(frame, 64, 148, 6, 340, "BROWN");

  // Headline: 92px Black (not italic)
  var hlColor; if (data.isCTA) { hlColor = "CREAM"; } else { hlColor = "BLACK"; }
  await addText(frame, data.headline, 90, 148, W - 154, 92, "Black", hlColor, "LEFT", 0.90);

  // Horizontal divider
  addRect(frame, 64, 730, W - 128, 2, "BROWN_L", 0.5);

  // Pills (if any)
  if (data.pills) {

    for (var j = 0; j < data.pills.length; j++) {
      var pill = data.pills[j];
      await figma.loadFontAsync({ family: "Inter", style: "Bold" });
      var pf = figma.createFrame();

      pt.resize(160, 36); pt.x = 0; pt.y = 12;
      pf.appendChild(pt);
      pillX += 172;
    }
  }

  // Subtext
  if (data.subtext) {

    var subColor;
    if (data.isCTA) { subColor = { r: 0.90, g: 0.87, b: 0.80 }; } else { subColor = { r: 0.25, g: 0.18, b: 0.10 }; }

async function main() {
  var frames = [];
  for (var i = 0; i < SLIDES.length; i++) {
    var frame = await buildSlide(SLIDES[i], i * (W + GAP));

  figma.viewport.scrollAndZoomIntoView(frames);
  figma.notify("Power Coffee - slides created!", { timeout: 4000 });
}
